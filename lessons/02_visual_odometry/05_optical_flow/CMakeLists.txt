if(CMAKE_HOST_APPLE)
  cmake_minimum_required(VERSION 3.20)
  include(../../../cmake/MacLesson.cmake)
  return()
endif()

cmake_minimum_required(VERSION 3.1)
project(optical_flow LANGUAGES CXX)

find_package(OpenCV REQUIRED COMPONENTS core imgproc imgcodecs highgui video OPTIONAL_COMPONENTS optflow)
set(flow_examples optical_flow)
if(OpenCV_VERSION VERSION_LESS 4 AND NOT TARGET opencv_optflow)
  message(STATUS "Install OpenCV 3 contrib optflow to build the DIS example")
else()
  list(APPEND flow_examples dis_optical_flow)
endif()
foreach(name IN LISTS flow_examples)
  add_executable(${name} src/${name}.cpp)
  set_target_properties(${name} PROPERTIES CXX_STANDARD 11 CXX_STANDARD_REQUIRED ON)
  target_include_directories(${name} PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../../../common" ${OpenCV_INCLUDE_DIRS})
  target_link_libraries(${name} PRIVATE ${OpenCV_LIBS})
endforeach()
